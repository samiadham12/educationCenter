<script>
        // State Management
        let currentMediaType = 'video';
        let selectedFile = null;

        const config = {
            video: { icon: 'movie', limit: '2GB', limitBytes: 2147483648, accept: 'video/*' },
            audio: { icon: 'audiotrack', limit: '500MB', limitBytes: 524288000, accept: 'audio/*' },
            pdf: { icon: 'description', limit: '50MB', limitBytes: 52428800, accept: '.pdf' }
        };

        // Tab Switching
        function switchTab(type) {
            currentMediaType = type;
            const tabs = ['video', 'audio', 'pdf'];
            tabs.forEach(t => {
                const btn = document.getElementById(`tab-${t}`);
                if (t === type) {
                    btn.classList.add('text-primary', 'border-b-2', 'border-primary');
                    btn.classList.remove('text-on-surface-variant');
                } else {
                    btn.classList.remove('text-primary', 'border-b-2', 'border-primary');
                    btn.classList.add('text-on-surface-variant');
                }
            });

            const uploadIcon = document.getElementById('upload-icon');
            const limitText = document.getElementById('size-limit-text');
            const fileInput = document.getElementById('file-input');

            uploadIcon.textContent = config[type].icon;
            limitText.textContent = `Maximum file size: ${config[type].limit}`;
            fileInput.accept = config[type].accept;
            
            resetUploadState();
        }

        // File Selection Logic
        const dropZone = document.getElementById('drop-zone');
        const fileInput = document.getElementById('file-input');
        const uploadStatus = document.getElementById('upload-status');

        dropZone.onclick = () => fileInput.click();

        fileInput.onchange = (e) => {
            const file = e.target.files[0];
            if (file) handleFile(file);
        };

        dropZone.ondragover = (e) => {
            e.preventDefault();
            dropZone.classList.add('bg-surface-container-highest');
        };

        dropZone.ondragleave = () => {
            dropZone.classList.remove('bg-surface-container-highest');
        };

        dropZone.ondrop = (e) => {
            e.preventDefault();
            dropZone.classList.remove('bg-surface-container-highest');
            const file = e.dataTransfer.files[0];
            if (file) handleFile(file);
        };

        function handleFile(file) {
            const limit = config[currentMediaType].limitBytes;
            if (file.size > limit) {
                alert(`File too large. Maximum size for ${currentMediaType} is ${config[currentMediaType].limit}`);
                return;
            }

            selectedFile = file;
            uploadStatus.classList.remove('hidden');
            document.getElementById('file-name').textContent = file.name;
            document.getElementById('file-size').textContent = (file.size / (1024 * 1024)).toFixed(2) + ' MB';
            document.getElementById('upload-progress-bar').style.width = '0%';
            document.getElementById('progress-percent').textContent = '0%';
            document.getElementById('upload-speed').textContent = 'Ready to upload';
        }

        function resetUploadState() {
            selectedFile = null;
            uploadStatus.classList.add('hidden');
            fileInput.value = '';
        }
