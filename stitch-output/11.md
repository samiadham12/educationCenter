        // Real-time Search Logic
        const searchInput = document.getElementById('lectureSearch');
        const lectureRows = document.querySelectorAll('.lecture-row');
        const cards = document.querySelectorAll('.glass-card');

        searchInput.addEventListener('input', (e) => {
            const term = e.target.value.toLowerCase();
            
            lectureRows.forEach(row => {
                const title = row.querySelector('.lecture-title')?.textContent.toLowerCase() || "";
                if (title.includes(term)) {
                    row.style.display = 'flex';
                } else {
                    row.style.display = 'none';
                }
            });

            // Handle card visibility based on matching lectures
            cards.forEach(card => {
                const visibleLectures = card.querySelectorAll('.lecture-row[style="display: flex;"]');
                const title = card.querySelector('h5').textContent.toLowerCase();
                
                if (visibleLectures.length > 0 || title.includes(term)) {
                    card.style.display = 'flex';
                } else if (term !== "") {
                    card.style.display = 'none';
                } else {
                    card.style.display = 'flex';
                }
            });
        });

        // Simple Loading Simulation
        window.addEventListener('load', () => {
            const root = document.getElementById('browserRoot');
            const skeleton = document.getElementById('loadingSkeleton');
            
            skeleton.classList.remove('hidden');
            root.classList.add('hidden');

            setTimeout(() => {
                skeleton.classList.add('hidden');
                root.classList.remove('hidden');
                root.classList.add('animate-in', 'fade-in', 'duration-700');
            }, 800);
        });
