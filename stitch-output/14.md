import React, { useEffect, useState, useRef } from 'react';

interface PdfViewerProps {
  mediaId: string;
  userEmail: string;
  onError?: (message: string) => void;
}

export const PdfViewer: React.FC<PdfViewerProps> = ({ mediaId, userEmail, onError }) => {
  const [blobUrl, setBlobUrl] = useState<string | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const containerRef = useRef<HTMLDivElement>(null);

  const fetchPdf = async () => {
    setIsLoading(true);
    setError(null);
    try {
      const res = await fetch(`/api/stream/pdf/${mediaId}`, { credentials: 'include' });
      if (!res.ok) throw new Error('Document could not be loaded');
      
      const blob = await res.blob();
      const url = URL.createObjectURL(blob);
      setBlobUrl(url);
    } catch (err) {
      const msg = err instanceof Error ? err.message : 'Unable to load PDF';
      setError(msg);
      onError?.(msg);
    } finally {
      setIsLoading(false);
    }
  };

  useEffect(() => {
    fetchPdf();
    return () => {
      if (blobUrl) {
        URL.revokeObjectURL(blobUrl);
      }
    };
  }, [mediaId]);

  // Anti-download keyboard shortcuts
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      // Prevent Ctrl+S, Ctrl+P, Cmd+S, Cmd+P
      if ((e.ctrlKey || e.metaKey) && (e.key === 's' || e.key === 'p')) {
        e.preventDefault();
        return false;
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, []);

  return (
    <div 
      ref={containerRef}
      className="relative w-full h-[800px] bg-surface-container-low rounded-xl border border-outline-variant shadow-xl overflow-hidden select-none flex flex-col"
      onContextMenu={(e) => e.preventDefault()}
    >
      {/* Header */}
      <div className="px-lg py-md border-b border-outline-variant/50 flex items-center justify-between bg-surface-container">
        <div className="flex items-center gap-md">
          <span className="material-icons text-primary">description</span>
          <h3 className="text-body-lg font-bold text-on-surface">Lecture Document</h3>
        </div>
        <div className="flex items-center gap-sm px-sm py-1 rounded-full bg-error/10 border border-error/20">
          <span className="material-icons text-error text-body-sm">lock</span>
          <span className="text-label-sm font-bold text-error uppercase tracking-tighter">Protected View</span>
        </div>
      </div>

      {/* Main Content Area */}
      <div className="relative flex-1 bg-surface-container-lowest overflow-hidden">
        {isLoading ? (
          <div className="absolute inset-0 flex flex-col items-center justify-center gap-md">
            <div className="w-12 h-12 border-4 border-primary border-t-transparent rounded-full animate-spin"></div>
            <p className="text-label-md text-on-surface-variant font-medium animate-pulse">Decrypting document...</p>
          </div>
        ) : error ? (
          <div className="absolute inset-0 flex flex-col items-center justify-center p-lg text-center">
            <span className="material-icons text-error text-display-md mb-md">error_outline</span>
            <p className="text-body-lg text-on-surface mb-lg font-medium">{error}</p>
            <button
              onClick={() => fetchPdf()}
              className="px-lg py-sm bg-primary text-on-primary rounded-full hover:bg-primary-container hover:text-on-primary-container transition-colors font-semibold"
            >
              Retry Loading
            </button>
          </div>
        ) : (
          <>
            {/* PDF Render */}
            <iframe 
              src={`${blobUrl}#toolbar=0&navpanes=0&scrollbar=0`}
              className="w-full h-full border-none"
              title="Protected Document Content"
            />

            {/* Security Overlays */}
            {/* Watermark Overlay */}
            <div className="absolute inset-0 pointer-events-none overflow-hidden flex flex-wrap content-around justify-around p-lg opacity-15">
              {[...Array(12)].map((_, i) => (
                <div 
                  key={i} 
                  className="text-headline-sm font-black text-on-surface whitespace-nowrap rotate-[-30deg] uppercase tracking-widest select-none"
                  style={{ textShadow: '1px 1px 0px rgba(0,0,0,0.1)' }}
                >
                  {userEmail}
                </div>
              ))}
            </div>

            {/* Interaction Shield */}
            <div className="absolute inset-0 pointer-events-none bg-[repeating-linear-gradient(-45deg,transparent,transparent_100px,rgba(255,255,255,0.02)_100px,rgba(255,255,255,0.02)_200px)]" />
          </>
        )}
      </div>

      {/* Footer Info */}
      <div className="px-lg py-sm bg-surface-container border-t border-outline-variant/50 flex items-center justify-center gap-sm">
        <span className="material-icons text-on-surface-variant text-body-sm">info</span>
        <p className="text-label-sm font-medium text-on-surface-variant">
          Printing and downloading are disabled to protect institutional material.
        </p>
      </div>

      {/* Print protection CSS */}
      <style>{`
        @media print {
          body * { display: none !important; }
          .no-print { display: none !important; }
        }
        iframe::-webkit-scrollbar {
          display: none;
        }
      `}</style>
    </div>
  );
};
