import React, { useEffect, useRef, useState } from 'react';
import Hls from 'hls.js';

interface AudioPlayerProps {
  lectureId: string;
  mediaId: string;
  title?: string;
  onError?: (message: string) => void;
}

interface ProgressData {
  lastPosition: number;
  watchedSeconds: number;
  totalDuration: number;
}

export const AudioPlayer: React.FC<AudioPlayerProps> = ({ lectureId, mediaId, title = "Lecture Audio", onError }) => {
  const audioRef = useRef<HTMLAudioElement>(null);
  const hlsRef = useRef<Hls | null>(null);
  const [isPlaying, setIsPlaying] = useState(false);
  const [playbackRate, setPlaybackRate] = useState<number>(1);
  const [currentTime, setCurrentTime] = useState(0);
  const [duration, setDuration] = useState(0);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [hasLoadedMetadata, setHasLoadedMetadata] = useState(false);

  // Load saved playback rate
  useEffect(() => {
    const savedRate = localStorage.getItem('audio-playback-rate');
    if (savedRate) {
      const rate = parseFloat(savedRate);
      setPlaybackRate(rate);
      if (audioRef.current) {
        audioRef.current.playbackRate = rate;
      }
    }
  }, []);

  const handleSpeedChange = (rate: number) => {
    setPlaybackRate(rate);
    localStorage.setItem('audio-playback-rate', rate.toString());
    if (audioRef.current) {
      audioRef.current.playbackRate = rate;
    }
  };

  const togglePlay = () => {
    if (audioRef.current) {
      if (isPlaying) {
        audioRef.current.pause();
      } else {
        audioRef.current.play();
      }
      setIsPlaying(!isPlaying);
    }
  };

  const formatTime = (seconds: number) => {
    const mins = Math.floor(seconds / 60);
    const secs = Math.floor(seconds % 60);
    return `${mins}:${secs.toString().padStart(2, '0')}`;
  };

  const fetchManifestAndProgress = async () => {
    setIsLoading(true);
    setError(null);
    try {
      // Step 1: Get manifest
      const manifestRes = await fetch(`/api/stream/${mediaId}/manifest`, { credentials: 'include' });
      if (!manifestRes.ok) throw new Error('Failed to load audio manifest');
      const { manifestUrl } = await manifestRes.json();

      // Step 3: Get progress
      const progressRes = await fetch(`/api/progress/${mediaId}`, { credentials: 'include' });
      let lastPosition = 0;
      if (progressRes.ok) {
        const { progress } = (await progressRes.json()) as { progress: ProgressData };
        lastPosition = progress.lastPosition;
      }

      // Step 2: Initialize Playback
      if (audioRef.current) {
        const audio = audioRef.current;

        if (Hls.isSupported()) {
          const hls = new Hls({
            xhrSetup: (xhr) => {
              xhr.withCredentials = true;
            },
          });
          hlsRef.current = hls;
          hls.loadSource(manifestUrl);
          hls.attachMedia(audio);

          hls.on(Hls.Events.MANIFEST_PARSED, () => {
            setIsLoading(false);
          });

          hls.on(Hls.Events.ERROR, (_, data) => {
            if (data.fatal) {
              setError('Unable to play audio');
              onError?.('Unable to play audio');
            }
          });
        } else if (audio.canPlayType('application/vnd.apple.mpegurl')) {
          audio.src = manifestUrl;
        }

        const onLoadedMetadata = () => {
          audio.currentTime = lastPosition;
          audio.playbackRate = playbackRate;
          setDuration(audio.duration);
          setHasLoadedMetadata(true);
          setIsLoading(false);
        };

        const onTimeUpdate = () => {
          setCurrentTime(audio.currentTime);
        };

        audio.addEventListener('loadedmetadata', onLoadedMetadata);
        audio.addEventListener('timeupdate', onTimeUpdate);
        
        return () => {
          audio.removeEventListener('loadedmetadata', onLoadedMetadata);
          audio.removeEventListener('timeupdate', onTimeUpdate);
        };
      }
    } catch (err) {
      const msg = err instanceof Error ? err.message : 'Unable to play audio';
      setError(msg);
      onError?.(msg);
      setIsLoading(false);
    }
  };

  useEffect(() => {
    fetchManifestAndProgress();
    return () => {
      if (hlsRef.current) {
        hlsRef.current.destroy();
      }
    };
  }, [mediaId]);

  // Step 4: Heartbeat
  useEffect(() => {
    const heartbeatInterval = setInterval(async () => {
      const audio = audioRef.current;
      if (audio && !audio.paused && !document.hidden && hasLoadedMetadata) {
        try {
          await fetch('/api/progress/heartbeat', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
              mediaId,
              lectureId,
              currentTimeSeconds: Math.floor(audio.currentTime),
              durationSeconds: Math.floor(audio.duration || 0),
              isPlaying: true,
            }),
          });
        } catch (err) {
          console.error('Heartbeat failed', err);
        }
      }
    }, 1000);

    return () => clearInterval(heartbeatInterval);
  }, [mediaId, lectureId, hasLoadedMetadata]);

  const handleSeek = (e: React.ChangeEvent<HTMLInputElement>) => {
    const time = parseFloat(e.target.value);
    if (audioRef.current) {
      audioRef.current.currentTime = time;
      setCurrentTime(time);
    }
  };

  return (
    <div 
      className="w-full bg-surface-container-low rounded-xl border border-outline-variant shadow-lg overflow-hidden select-none"
      onContextMenu={(e) => e.preventDefault()}
    >
      <div className="p-lg">
        {/* Header */}
        <div className="flex items-center gap-md mb-lg">
          <div className="w-12 h-12 rounded-lg bg-primary-container/20 flex items-center justify-center">
            <span className="material-icons text-primary text-headline-md">audiotrack</span>
          </div>
          <div>
            <h3 className="text-headline-sm font-bold text-on-surface line-clamp-1">{title}</h3>
            <p className="text-label-md text-on-surface-variant font-medium">Lecture Audio Resource</p>
          </div>
        </div>

        {/* Hidden Audio Element */}
        <audio 
          ref={audioRef} 
          preload="metadata" 
          controlsList="nodownload"
          onPlay={() => setIsPlaying(true)}
          onPause={() => setIsPlaying(false)}
          className="hidden"
        />

        {/* Playback Controls Area */}
        <div className="bg-surface-container-highest rounded-xl p-md mb-lg border border-outline-variant/50">
          <div className="flex items-center gap-lg">
            {/* Play/Pause Button */}
            <button
              onClick={togglePlay}
              disabled={isLoading || !!error}
              className="w-14 h-14 rounded-full bg-primary text-on-primary flex items-center justify-center hover:scale-105 active:scale-95 transition-all shadow-md disabled:opacity-50"
            >
              <span className="material-icons text-display-xs">
                {isPlaying ? 'pause' : 'play_arrow'}
              </span>
            </button>

            {/* Seek Bar and Waveform Container */}
            <div className="flex-1 flex flex-col gap-sm">
              {/* CSS Waveform (Simple animated bars) */}
              <div className="flex items-end gap-0.5 h-8 px-1 mb-1">
                {[...Array(40)].map((_, i) => (
                  <div 
                    key={i} 
                    className={`flex-1 bg-primary/30 rounded-t-full transition-all duration-300 ${isPlaying ? 'animate-pulse' : ''}`}
                    style={{ 
                      height: isPlaying ? `${20 + Math.random() * 80}%` : '20%',
                      animationDelay: `${i * 0.05}s`
                    }}
                  />
                ))}
              </div>

              {/* Range Input */}
              <input
                type="range"
                min="0"
                max={duration || 0}
                value={currentTime}
                onChange={handleSeek}
                className="w-full h-1.5 bg-surface-variant rounded-full appearance-none cursor-pointer accent-primary"
              />

              {/* Time Display */}
              <div className="flex justify-between items-center mt-1">
                <span className="text-label-sm font-bold text-primary">{formatTime(currentTime)}</span>
                <span className="text-label-sm font-medium text-on-surface-variant">{formatTime(duration)}</span>
              </div>
            </div>
          </div>
        </div>

        {/* Speed Controls and Status Row */}
        <div className="flex flex-col md:flex-row items-center justify-between gap-md border-t border-outline-variant/30 pt-md">
          <div className="flex items-center gap-sm">
            <span className="text-label-sm text-on-surface-variant font-bold uppercase tracking-wider">Speed</span>
            <div className="flex bg-surface-container-low rounded-lg p-1 border border-outline-variant/50">
              {[1, 1.25, 1.5, 2, 2.5, 3].map((rate) => (
                <button
                  key={rate}
                  onClick={() => handleSpeedChange(rate)}
                  className={`
                    px-3 py-1 text-label-sm font-bold rounded-md transition-all
                    ${playbackRate === rate 
                      ? 'bg-primary text-on-primary shadow-sm' 
                      : 'text-on-surface-variant hover:text-on-surface hover:bg-surface-variant'
                    }
                  `}
                >
                  {rate}x
                </button>
              ))}
            </div>
          </div>

          <div className="flex items-center gap-sm text-emerald-500">
            <span className="material-icons text-body-sm">check_circle</span>
            <span className="text-label-sm font-bold">Progress saved</span>
          </div>
        </div>
      </div>

      {/* Overlays */}
      {isLoading && (
        <div className="absolute inset-0 bg-surface-container-low/60 backdrop-blur-[1px] flex items-center justify-center z-10">
          <div className="w-8 h-8 border-3 border-primary border-t-transparent rounded-full animate-spin"></div>
        </div>
      )}

      {error && (
        <div className="p-md bg-error/10 border-t border-error/20 flex items-center justify-center gap-md">
          <span className="material-icons text-error text-body-md">error_outline</span>
          <p className="text-label-md text-error font-bold">{error}</p>
          <button 
            onClick={() => fetchManifestAndProgress()}
            className="text-label-md font-black underline hover:text-error-container"
          >
            Retry
          </button>
        </div>
      )}
    </div>
  );
};
