        document.getElementById('redemption-form').addEventListener('submit', function(e) {
            e.preventDefault();
            
            const input = document.getElementById('lecture-code');
            const btn = document.getElementById('submit-btn');
            const btnText = document.getElementById('btn-text');
            const spinner = document.getElementById('loading-spinner');
            const errorBox = document.getElementById('error-box');
            const errorMsg = document.getElementById('error-message');
            
            // Validation
            if (input.value.length < 12) {
                errorMsg.innerText = "Please enter the full 12-character code.";
                errorBox.classList.remove('hidden');
                input.focus();
                return;
            }

            // Simulated Loading State
            errorBox.classList.add('hidden');
            btn.disabled = true;
            btnText.innerText = "Unlocking...";
            spinner.classList.remove('hidden');

            // Simulate API Call
            setTimeout(() => {
                const code = input.value;
                if (code === 'ERROR-CODE-XX') {
                    // Fail state
                    btn.disabled = false;
                    btnText.innerText = "Unlock Lecture";
                    spinner.classList.add('hidden');
                    errorMsg.innerText = "This code has already been redeemed or is expired.";
                    errorBox.classList.remove('hidden');
                } else {
                    // Success State
                    btnText.innerText = "Success! Redirecting...";
                    btn.classList.replace('bg-primary-container', 'bg-emerald-500');
                    setTimeout(() => {
                        window.location.href = '/student/watch/lecture-101';
                    }, 800);
                }
            }, 1500);
        });

        // Auto-formatting for XXXX-XXXX-XXXX pattern (Optional UX)
        document.getElementById('lecture-code').addEventListener('input', function(e) {
            let val = e.target.value.replace(/[^A-Z0-9]/gi, '').toUpperCase();
            if (val.length > 12) val = val.slice(0, 12);
            
            // Visual separation logic if needed, but the requirement asked for large monospace text-center
            // So we'll keep the value clean for functional validation
            e.target.value = val;
        });
