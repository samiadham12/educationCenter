        // UI Logic
        const userRole = "SUPER_ADMIN"; // Mock role from context

        async function initDashboard() {
            // Role Guard
            if (userRole !== "SUPER_ADMIN" && userRole !== "ADMIN") {
                document.getElementById('promotionAction').classList.add('hidden');
            }

            try {
                const data = await api.get('/api/admin/stats');
                renderStats(data);
                renderTable(data.logs);
            } catch (err) {
                document.getElementById('errorBanner').classList.remove('hidden');
            }
        }

        function renderStats(data) {
            document.getElementById('stat-total-students').innerText = data.totalStudents.toLocaleString();
            document.getElementById('stat-active-subs').innerText = data.activeSubs.toLocaleString();
            document.getElementById('stat-media-ready').innerText = data.mediaReady.toLocaleString();
            document.getElementById('stat-uploads').innerText = data.uploadsThisWeek.toLocaleString();
            document.getElementById('stat-uploads-subtitle').innerText = `${data.mediaReady} total items currently indexed`;
        }

        function renderTable(logs) {
            const tbody = document.getElementById('activityTableBody');
            tbody.innerHTML = logs.map(log => {
                const methodColors = {
                    'GET': 'bg-outline-variant/30 text-on-surface-variant',
                    'POST': 'bg-secondary-container/40 text-secondary',
                    'DELETE': 'bg-error-container/40 text-error'
                };
                return `
                    <tr class="border-b border-outline-variant/30 hover:bg-surface-container-high transition-colors">
                        <td class="px-lg py-md font-body-sm text-on-surface">${log.action}</td>
                        <td class="px-lg py-md">
                            <span class="px-sm py-xs rounded font-mono text-[10px] font-bold ${methodColors[log.method] || 'bg-outline-variant'}">${log.method}</span>
                        </td>
                        <td class="px-lg py-md text-on-surface-variant font-label-sm">${log.status}</td>
                        <td class="px-lg py-md text-right font-body-sm text-on-surface-variant opacity-60">${log.time}</td>
                    </tr>
                `;
            }).join('');
        }

        function retryFetch() {
            document.getElementById('errorBanner').classList.add('hidden');
            initDashboard();
        }

        // Modal/Toast interactions
        function showConfirmModal() {
            document.getElementById('confirmModal').classList.remove('hidden');
        }

        function hideConfirmModal() {
            document.getElementById('confirmModal').classList.add('hidden');
        }

        async function executePromotion() {
            const btn = document.getElementById('confirmExecuteBtn');
            const originalText = btn.innerText;
            btn.disabled = true;
            btn.innerHTML = `<span class="material-symbols-outlined animate-spin">sync</span> Running...`;
            
            try {
                const res = await api.post('/api/academic-years/promote-all');
                hideConfirmModal();
                showToast(res.message, "check_circle");
            } catch (err) {
                showToast("Operation failed. Please check logs.", "error");
            } finally {
                btn.disabled = false;
                btn.innerText = originalText;
            }
        }
